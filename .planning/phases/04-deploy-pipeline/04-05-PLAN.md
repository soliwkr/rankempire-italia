    **3. Deploy handler — logica completa:**

    ```typescript
    api.post('/:id/deploy', async (c) => {
      const db = drizzle(c.env.DB);
      const id = c.req.param('id');

      // Validazione input id (prevenire injection)
      if (!/^[0-9a-f-]{36}$/.test(id)) {
        return c.json({ error: 'Invalid project id format' }, 400);
      }

      // Fetch progetto
      const project = await db.select().from(projects).where(eq(projects.id, id)).get();
      if (!project) {
        return c.json({ error: 'Project not found' }, 404);
      }

      // Idempotency check (D-12)
      if (project.status === 'live') {
        return c.json({
          error: 'Project already deployed',
          pages_url: project.pagesUrl,
          github_repo_url: project.githubRepoUrl,
        }, 409);
      }

      // Validazione slug (prevenire injection nel nome repo/pages)
      if (!/^[a-z0-9-]+$/.test(project.slug)) {
        return c.json({ error: 'Invalid project slug format' }, 422);
      }

      const body = await c.req.json().catch(() => ({}));

      // Inizializza services
      const github = new GitHubService({
        token: c.env.GITHUB_TOKEN,
        templateOwner: c.env.GITHUB_TEMPLATE_OWNER,
        templateRepo: c.env.GITHUB_TEMPLATE_REPO,
      });
      const cfPages = new CloudflarePagesService({
        apiToken: c.env.CF_API_TOKEN,
        accountId: c.env.CF_ACCOUNT_ID,
      });

      try {
        const repoName = `rr-${project.slug}`;
        let repoOwner: string;
        let repoGithubName: string;

        // Step 1: Crea repo GitHub (skip se già completato)
        if (project.status === 'pending') {
          const repoResponse = await github.createProjectRepo(
            repoName,
            `Rank & Rent Site for ${project.niche} in ${project.location}`
          );
          // createProjectRepo ora include il retry loop internamente
          // se il repo non è pronto dopo 10 tentativi, lancia un Error
          repoOwner = (repoResponse as any).owner?.login ?? c.env.GITHUB_TEMPLATE_OWNER;
          repoGithubName = (repoResponse as any).name ?? repoName;

          await db.update(projects)
            .set({
              status: 'repo_created',
              githubRepoUrl: `https://github.com/${repoOwner}/${repoGithubName}`,
            })
            .where(eq(projects.id, id));
        } else {
          // Riprendi da status parziale: estrai owner/name dal githubRepoUrl salvato
          const savedUrl = project.githubRepoUrl ?? '';
          const urlParts = savedUrl.replace('https://github.com/', '').split('/');
          repoOwner = urlParts[0] ?? c.env.GITHUB_TEMPLATE_OWNER;
          repoGithubName = urlParts[1] ?? repoName;
        }

        // Step 2: Inietta site.config.json (D-07/D-08)
        if (project.status === 'pending' || project.status === 'repo_created') {
          const siteConfig = {
            projectId: project.id,
            niche: project.niche,
            city: project.location,
            services: (body as any).services ?? [],
            zones: (body as any).zones ?? [],
            avatar: (body as any).avatar ?? 'in-pain',
            factoryApi: c.env.FACTORY_API_URL,
            slug: project.slug,
            domain: null,
            gmbPlaceId: null,
            ga4MeasurementId: null,
            businessName: null,
            phone: null,
            address: null,
            geo: null,
          };

          await github.createFile(
            repoOwner,
            repoGithubName,
            'src/data/site.config.json',
            JSON.stringify(siteConfig, null, 2),
            'Inject site configuration'
          );

          await db.update(projects)
            .set({ status: 'pages_linked' })
            .where(eq(projects.id, id));
        }

        // Step 3: Crea CF Pages project (D-01/D-02/D-03/D-04)
        const pagesResult = await cfPages.createProject(
          project.slug,
          repoOwner,
          repoGithubName,
          c.env.FACTORY_API_URL
        );

        const pagesUrl = `https://${(pagesResult as any).subdomain ?? `rr-${project.slug}`}.pages.dev`;
        const pagesProjectName = (pagesResult as any).name ?? `rr-${project.slug}`;

        await db.update(projects)
          .set({
            status: 'deploying',
            pagesProjectName,
            pagesUrl,
          })
          .where(eq(projects.id, id));

        return c.json({
          success: true,
          repoUrl: `https://github.com/${repoOwner}/${repoGithubName}`,
          pagesUrl,
          status: 'deploying',
        });

      } catch (err: any) {
        // Se il retry loop di GitHub è esaurito, err.message contiene "not ready after 10 attempts"
        if (err.message?.includes('not ready after 10 attempts')) {
          // Status già salvato come repo_created nel try block
          return c.json({
            error: 'GitHub repo not ready after 10 attempts — richiama /deploy per riprendere',
            status: 'repo_created',
          }, 503);
        }
        console.error('[deploy] Error:', err.message);
        return c.json({ error: 'Deployment failed', details: err.message }, 500);
      }
    });
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP request body → deploy endpoint | services[], zones[], avatar vengono dal client — non validati strutturalmente in Phase 4 |
| CF_API_TOKEN in handler → CF API | Token Cloudflare global con Pages:Edit — se esposto, attaccante crea Pages projects arbitrari |
| project.slug → GitHub API / CF Pages API | Slug dal D1 database — se contenesse caratteri speciali, potrebbe iniettarsi in nomi repo/project |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-05-01 | Injection | project.slug → nome repo/Pages project | mitigate | Validazione `/^[a-z0-9-]+$/` su slug PRIMA di passarlo a GitHubService e CloudflarePagesService. Se formato non valido → 422. Acceptance criteria include grep per validazione slug |
| T-04-05-02 | Injection | id param → WHERE clause D1 | mitigate | Validazione UUID format `/^[0-9a-f-]{36}$/` su `id` prima di qualsiasi query D1. Acceptance criteria include grep per validazione UUID |
| T-04-05-03 | Information Disclosure | CF_API_TOKEN in error log | mitigate | Il catch block usa solo `err.message` — mai `c.env.CF_API_TOKEN` in log o response. `console.error('[deploy] Error:', err.message)` — senza token. Accettato: logs CF worker visibili solo all'operatore |
| T-04-05-04 | Elevation of Privilege | Deploy senza auth | accept | Il deploy endpoint è montato su `protectedApp` in index.ts — bearerAuth middleware applicato automaticamente. Nessuna modifica richiesta. Verificare che `protectedApp.route('/api/projects', projectsApi)` rimanga invariato in index.ts |
| T-04-05-05 | Spoofing | Replay attack su deploy endpoint | accept | L'idempotency check (409 se già 'live') previene deploy multipli per lo stesso progetto. Cloudflare Pages API ritorna 409 se Pages project con quel nome esiste già — gestito dal catch block |
| T-04-05-06 | Tampering | site.config.json con body arbitrario | accept | services/zones/avatar dal body non sono validati strutturalmente in Phase 4 (solo fallback a [] / 'in-pain'). Rischio basso: l'attaccante avrebbe già dovuto passare bearerAuth. Struttura JSON scritta via GitHub API (non eseguita server-side) |
